        // TAB SWITCHER
        function switchTab(tabId) {
            document.querySelectorAll('.tab-content').forEach(el => el.classList.add('hidden'));
            document.querySelectorAll('.nav-btn').forEach(el => {
                el.classList.remove('bg-brand-500', 'text-white');
                el.classList.add('text-gray-600');
            });

            const activeTab = document.getElementById(`section-${tabId}`);
            if (activeTab) activeTab.classList.remove('hidden');

            const activeNav = document.getElementById(`nav-${tabId}`);
            if (activeNav) {
                activeNav.classList.add('bg-brand-500', 'text-white');
                activeNav.classList.remove('text-gray-600');
            }
        }

        function filterAsas(category) {
            document.querySelectorAll('.filter-asas-btn').forEach(btn => {
                btn.classList.remove('bg-brand-500', 'text-white');
                btn.classList.add('bg-white', 'text-gray-600');
            });
            event.target.classList.add('bg-brand-500', 'text-white');
            event.target.classList.remove('bg-white', 'text-gray-600');

            if (category === 'all') {
                renderAsas9Cards(dataAsas9);
            } else {
                renderAsas9Cards(dataAsas9.filter(x => x.category === category));
            }
        }

        function renderActionPlanTable() {
            const tbody = document.getElementById('action-plan-tbody');
            tbody.innerHTML = actionPlanRows.map((row, idx) => `
                <tr>
                    <td class="p-2.5"><input type="text" value="${row.asas}" onchange="updateActionRow(${idx}, 'asas', this.value)" class="w-full p-2 border border-gray-300 rounded-lg text-xs"></td>
                    <td class="p-2.5"><input type="text" value="${row.solusi}" onchange="updateActionRow(${idx}, 'solusi', this.value)" class="w-full p-2 border border-gray-300 rounded-lg text-xs"></td>
                    <td class="p-2.5"><input type="text" value="${row.indikator}" onchange="updateActionRow(${idx}, 'indikator', this.value)" class="w-full p-2 border border-gray-300 rounded-lg text-xs"></td>
                    <td class="p-2.5 text-center"><button onclick="deleteActionRow(${idx})" class="text-red-500 hover:text-red-700 font-bold">✕</button></td>
                </tr>
            `).join('');
        }

        function addActionButton() {
            actionPlanRows.push({ asas: "Asas ...", solusi: "Solusi baru...", indikator: "Indikator..." });
            renderActionPlanTable();
        }

        function deleteActionRow(idx) {
            actionPlanRows.splice(idx, 1);
            renderActionPlanTable();
        }

        function updateActionRow(idx, field, val) {
            actionPlanRows[idx][field] = val;
        }

        // RUBRIC SCORING CALCULATOR
        function updateRubricScore() {
            const s1 = parseInt(document.getElementById('rubric-score-1').value) || 0;
            const s2 = parseInt(document.getElementById('rubric-score-2').value) || 0;
            const s3 = parseInt(document.getElementById('rubric-score-3').value) || 0;
            const s4 = parseInt(document.getElementById('rubric-score-4').value) || 0;
            const total = s1 + s2 + s3 + s4;
            document.getElementById('rubric-total-score').innerText = `Total Skor: ${total} / 100`;
        }
